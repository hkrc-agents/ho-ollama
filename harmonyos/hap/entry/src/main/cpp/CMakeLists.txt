cmake_minimum_required(VERSION 3.16)
project(OllamaHarmonyBridge LANGUAGES CXX)
option(PICO_MANAGED_NATIVE_SERVICES "Use official Native SO service creation" OFF)
option(PICO_ELF_SUPERVISION "Use parent lifeline and scoped subreaper for executable services" OFF)
if(PICO_MANAGED_NATIVE_SERVICES AND PICO_ELF_SUPERVISION)
  message(FATAL_ERROR "Select SO or ELF supervision, never both")
endif()
if(PICO_MANAGED_NATIVE_SERVICES)
  if(NOT IS_DIRECTORY "${PICO_NATIVE_SERVICES_SOURCE}" OR NOT IS_DIRECTORY "${PICO_JSON_INCLUDE}")
    message(FATAL_ERROR "Explicit native-services and JSON include sources required")
  endif()
  set(BUILD_TESTING OFF CACHE BOOL "" FORCE)
  add_subdirectory("${PICO_NATIVE_SERVICES_SOURCE}" "${CMAKE_CURRENT_BINARY_DIR}/native-services")
  add_library(ollamabridge SHARED bridge_managed.cpp storage_permission_audit.cpp)
  target_include_directories(ollamabridge PRIVATE "${PICO_JSON_INCLUDE}")
  target_link_libraries(ollamabridge PRIVATE pico_service_ohos)
else()
  add_library(ollamabridge SHARED bridge.cpp storage_permission_audit.cpp)
endif()
target_compile_features(ollamabridge PRIVATE cxx_std_17)
if(PICO_ELF_SUPERVISION)
  target_compile_definitions(ollamabridge PRIVATE PICO_ELF_SUPERVISION=1)
  set(PICO_ELF_SOURCE_PARTS "")
  foreach(part bridge.cpp elf_supervision.h storage_permission_audit.cpp storage_permission_audit.h CMakeLists.txt)
    set_property(DIRECTORY APPEND PROPERTY CMAKE_CONFIGURE_DEPENDS "${CMAKE_CURRENT_SOURCE_DIR}/${part}")
    file(SHA256 "${CMAKE_CURRENT_SOURCE_DIR}/${part}" part_sha)
    string(APPEND PICO_ELF_SOURCE_PARTS "${part_sha}")
  endforeach()
  string(SHA256 PICO_ELF_SOURCE_SHA256 "${PICO_ELF_SOURCE_PARTS}")
  target_compile_definitions(ollamabridge PRIVATE PICO_ELF_SOURCE_SHA256="${PICO_ELF_SOURCE_SHA256}")
endif()
if(CMAKE_BUILD_TYPE STREQUAL "Debug")
  target_compile_definitions(ollamabridge PRIVATE PICO_ENABLE_DEBUG_COMMANDS=1)
else()
  target_compile_definitions(ollamabridge PRIVATE PICO_ENABLE_DEBUG_COMMANDS=0)
endif()
target_compile_options(ollamabridge PRIVATE -Wall -Wextra -Werror -Wno-unused-command-line-argument)
target_link_libraries(ollamabridge PRIVATE libace_napi.z.so libdl.so)
